Algoritmo cuantosAniosTiene
#definir tipo de datos
	DEFINIR DA, MA, AA, DN, MN, AN, edad Como Entero
	
#leer datos por teclado
	ESCRIBIR "Dime cuando naciste y te dire cuantos años tienes. Primero el dia que naciste: "
	LEER DN
	ESCRIBIR "Segundo el mes: "
	LEER MN
	ESCRIBIR "Tercero el año: "
	LEER AN
	
		
#proceso y condicional	
	edad = AA - AN
	SI ((MN > MA) OR ((MN == MA) AND (DN > DA))  ENTONCES
		edad = edad - 1
	FINSI 
	
#imprimir resultado
	ESCRIBIR "Tienes " + edad + " años."
	
FinAlgoritmo

algoritmo sacar_dinero
#definir tipo de datos
	DEFINIR sacarDinero, saldo, saldoRestante Como Real

#Declarar variables
	saldo = 500	

#leer datos por teclado
	ESCRIBIR "Tu saldo disponible es de %.2f €.",saldo
	ESCRIBIR "Dime la cantidad de dinero que quieres retirar: "
	LEER sacarDinero

#proceso, condicional e imprimir resultado	
	
	SI saldo<sacarDinero ENTONCES
		ESCRIBIR "Cantidad no valida. Tu saldo es de %.2f €.",saldo
	FINSI
	SINO 
		saldoRestante = saldo - sacarDinero
		ESCRIBIR "Tu saldo restante es de %.2f €",saldoRestante
	FINSI
