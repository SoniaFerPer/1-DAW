ALGORITMO esPrimo
#definir tipo de datos
	DEFINIR num Como Entero
	
#leer los datos por teclado	
	ESCRIBIR "Dime un numero: "
	LEER num
	
#proceso, condicion e imprimir resultado	
	SI num == 2 OR num == 3 OR num == 5 OR num == 7 ENTONCES
		ESCRIBIR "Es un número primo."
	SINO SI num MOD 2 == 0 OR num MOD 3 == 0 OR num MOD 5 == 0 OR num MOD 7 == 0 ENTONCES 
		ESCRIBIR "No es un número primo."
	SINO
		ESCRIBIR "Es un número primo."
	FINSI
FINALGORITMO

ALGORITMO triangulo
#definir tipo de datos
	DEFINIR grado1, grado2, grado3 Como Entero

#leer los datos por teclado
	ESCRIBIR "Introduce el primer ángulo del triangulo: "
	LEER grado1
	ESCRIBIR "Introduce el segundo ángulo del triangulo: "
	LEER grado2
	ESCRIBIR "Introduce el tercer ángulo del triangulo: "
	LEER grado3
	
#proceso, condicion e imprimir resultado
	SI grado1 + grado2 + grado3 > 180 ENTONCES
        ESCRIBIR "No puede ser un triangulo."
    SINO SI grado1 < 90 AND grado2 < 90 AND grado3 < 90 ENTONCES
        ESCRIBIR "El triangulo es acutángulo."
    SINO SI (grado1 == 90 OR grado2 == 90 OR grado3 == 90) AND (grado1 + grado2 == 90 OR grado1 + grado3 == 90 OR grado2 + grado3 == 90 ENTONCES
        ESCRIBIR "El triangulo es rectángulo."
    SINO SI grado1 > 90 OR grado2 > 90 OR grado3 > 90 ENTONCES
        ESCRIBIR ""El triangulo es obtusángulo." 
	FINSI
FINALGORITMO

#definir tipo de datos
	DEFINIR resultado1, resultado2, a, b, c, discriminante Como Real

#leer los datos por teclado
	ESCRIBIR "valor de a: "
	LEER a
	ESCRIBIR "valor de b: "
	LEER b
	ESCRIBIR "valor de c: "
	LEER c

#proceso, condicion e imprimir resultado
	discriminante = Math.pow(b,2) - 4 * a * c

	SI discriminante>0 ENTONCES
        ESCRIBIR "La ecuación tiene dos soluciones distintas: "
		resultado1 = (-b + Math.sqrt(discriminante)) / (2 * a)
        resultado2 = (-b - Math.sqrt(discriminante)) / (2 * a)
		ESCRIBIR "Primer resultado: " + resultado1
		ESCRIBIR "Segundo resultado: " + resultado2
    SINO SI discriminante == 0 ENTONCES
		resultado1 = -b / (2 * a);
        ESCRIBIR "La ecuación tiene una única solución (raíz doble): " + resultado1
    SINO
        ESCRIBIR "No existe una solución real."
    
	FINSI
